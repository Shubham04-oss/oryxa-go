44// retrySchedule is how long to wait before each retry of a failed invoice.
45var retrySchedule = []time.Duration{
46 1 * time.Hour,
47 6 * time.Hour,
48 24 * time.Hour,
49}
50
51// RetryInvoice charges the invoice again, up to len(retrySchedule) times.
52func (s *Service) RetryInvoice(ctx context.Context, inv *Invoice) error {
53 if inv.Attempts >= len(retrySchedule) {
54 return s.markUncollectible(ctx, inv)
55 }
56 wait := retrySchedule[inv.Attempts]
57 return s.queue.Schedule(ctx, inv.ID, time.Now().Add(wait))
58}
59
60// markUncollectible stops retries and tells the customer's admins.
61func (s *Service) markUncollectible(ctx context.Context, inv *Invoice) error {
62 inv.Status = StatusUncollectible
63 if err := s.store.Save(ctx, inv); err != nil {
64 return err
65 }
66 return s.notify.Admins(ctx, inv.OrgID, "invoice_uncollectible", inv)
67}
